cmake_minimum_required(VERSION 3.24)
project(Aetheri VERSION 0.0.1 LANGUAGES C)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Aetheri native components target Linux. Host preparation tools are separate.")
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(AETHERI_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(AETHERI_PRIVILEGED_TESTS "Run privileged probes in the disposable guest" OFF)
option(AETHERI_LINUX_IPC "Build the native sd-bus peer integration" OFF)
option(AETHERI_POLICY_PROBES "Build the disposable SELinux policy probe (does not install policy)" OFF)

find_package(OpenSSL 3.5.9 REQUIRED COMPONENTS Crypto)
if(NOT OPENSSL_VERSION VERSION_LESS "3.6.0")
    message(FATAL_ERROR "Review a new OpenSSL release family before adoption; supported family is 3.5 LTS.")
endif()
add_library(aetheri_core STATIC src/core/owner.c src/core/storage.c src/core/fixture.c src/core/root.c src/core/allocation.c src/core/digest.c src/core/import.c src/core/publish.c)
target_include_directories(aetheri_core PUBLIC src)
target_link_libraries(aetheri_core PUBLIC OpenSSL::Crypto)
target_compile_definitions(aetheri_core PRIVATE _GNU_SOURCE)
target_compile_options(aetheri_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(AETHERI_SANITIZERS)
    target_compile_options(aetheri_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(aetheri_core PUBLIC -fsanitize=address,undefined)
endif()

find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(SQLite3 3.37 REQUIRED)
if(NOT TARGET SQLite3::SQLite3)
    add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
set(AETHERI_SCHEMA_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema.c")
add_custom_command(OUTPUT "${AETHERI_SCHEMA_SOURCE}"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_schema.py" "${AETHERI_SCHEMA_SOURCE}"
    DEPENDS tools/embed_schema.py tools/catalogue_contracts.py
            spec/schema/common-v2.sql spec/schema/system-v2.sql spec/schema/user-v2.sql
    VERBATIM)
add_library(aetheri_catalogue STATIC src/catalogue/store.c "${AETHERI_SCHEMA_SOURCE}")
target_include_directories(aetheri_catalogue PUBLIC src)
target_compile_definitions(aetheri_catalogue PRIVATE _GNU_SOURCE)
target_compile_options(aetheri_catalogue PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
target_link_libraries(aetheri_catalogue PUBLIC aetheri_core SQLite3::SQLite3)
