cmake_minimum_required(VERSION 3.24)
project(Aetheri VERSION 0.0.1 LANGUAGES C)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Aetheri native components target Linux. Host preparation tools are separate.")
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(AETHERI_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(AETHERI_PRIVILEGED_TESTS "Run private mount-namespace probes in the disposable guest" OFF)

add_library(aetheri_core STATIC src/core/owner.c src/core/storage.c)
target_include_directories(aetheri_core PUBLIC src)
target_compile_definitions(aetheri_core PRIVATE _GNU_SOURCE)
target_compile_options(aetheri_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(AETHERI_SANITIZERS)
    target_compile_options(aetheri_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(aetheri_core PUBLIC -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.c)
    target_compile_definitions(test_core PRIVATE _GNU_SOURCE)
    target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(test_core PRIVATE aetheri_core)
    add_test(NAME core.owner COMMAND test_core owner "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME core.storage COMMAND test_core storage "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(core.owner core.storage PROPERTIES TIMEOUT 10)
    if(AETHERI_PRIVILEGED_TESTS)
        add_test(NAME core.mount_containment COMMAND sudo -n unshare --mount --propagation private
                 $<TARGET_FILE:test_core> mount "${CMAKE_CURRENT_BINARY_DIR}")
        set_tests_properties(core.mount_containment PROPERTIES TIMEOUT 10)
    endif()
endif()
