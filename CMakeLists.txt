cmake_minimum_required(VERSION 3.24)
project(Aetheri VERSION 0.0.1 LANGUAGES C)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Aetheri native components target Linux. Host preparation tools are separate.")
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(AETHERI_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(AETHERI_PRIVILEGED_TESTS "Run privileged probes in the disposable guest" OFF)
option(AETHERI_LINUX_IPC "Build the native sd-bus peer integration" OFF)
option(AETHERI_POLICY_PROBES "Build the disposable SELinux policy probe (does not install policy)" OFF)

add_library(aetheri_core STATIC src/core/owner.c src/core/storage.c src/core/fixture.c)
target_include_directories(aetheri_core PUBLIC src)
target_compile_definitions(aetheri_core PRIVATE _GNU_SOURCE)
target_compile_options(aetheri_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(AETHERI_SANITIZERS)
    target_compile_options(aetheri_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(aetheri_core PUBLIC -fsanitize=address,undefined)
endif()

find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(SQLite3 3.37 REQUIRED)
if(NOT TARGET SQLite3::SQLite3)
    add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
set(AETHERI_SCHEMA_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema.c")
add_custom_command(OUTPUT "${AETHERI_SCHEMA_SOURCE}"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_schema.py" "${AETHERI_SCHEMA_SOURCE}"
    DEPENDS tools/embed_schema.py tools/catalogue_contracts.py
            spec/schema/common-v2.sql spec/schema/system-v2.sql spec/schema/user-v2.sql
    VERBATIM)
add_library(aetheri_catalogue STATIC src/catalogue/store.c "${AETHERI_SCHEMA_SOURCE}")
target_include_directories(aetheri_catalogue PUBLIC src)
target_compile_definitions(aetheri_catalogue PRIVATE _GNU_SOURCE)
target_compile_options(aetheri_catalogue PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
target_link_libraries(aetheri_catalogue PUBLIC aetheri_core SQLite3::SQLite3)

if(AETHERI_LINUX_IPC)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd>=259)
    pkg_check_modules(SELINUX REQUIRED IMPORTED_TARGET libselinux>=3.10)
    add_library(aetheri_linux STATIC src/linux/peer.c src/linux/process.c)
    target_include_directories(aetheri_linux PUBLIC src)
    target_compile_definitions(aetheri_linux PRIVATE _GNU_SOURCE)
    target_compile_options(aetheri_linux PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(aetheri_linux PUBLIC aetheri_core PkgConfig::SYSTEMD PkgConfig::SELINUX)
    set(AETHERI_WIRE_TABLES "${CMAKE_CURRENT_BINARY_DIR}/generated/wire-tables.h")
    add_custom_command(OUTPUT "${AETHERI_WIRE_TABLES}"
        COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_api.py" "${AETHERI_WIRE_TABLES}"
        DEPENDS tools/embed_api.py spec/api/management-v1.xml spec/api/documents-v1.xml spec/api/runtime-v1.xml spec/api/broker-v1.xml
        VERBATIM)
    add_executable(aetheri-authority src/authority/main.c src/authority/broker.c "${AETHERI_WIRE_TABLES}")
    target_include_directories(aetheri-authority PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(aetheri-authority PRIVATE _GNU_SOURCE)
    target_compile_options(aetheri-authority PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(aetheri-authority PRIVATE aetheri_catalogue aetheri_linux PkgConfig::SELINUX)
    if(AETHERI_POLICY_PROBES)
        add_executable(probe_service tests/probe_service.c)
        target_compile_definitions(probe_service PRIVATE _GNU_SOURCE)
        target_compile_options(probe_service PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        target_link_libraries(probe_service PRIVATE aetheri_linux PkgConfig::SELINUX)
        add_executable(probe_broker tests/probe_broker.c)
        target_compile_definitions(probe_broker PRIVATE _GNU_SOURCE)
        target_compile_options(probe_broker PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        target_link_libraries(probe_broker PRIVATE aetheri_linux)
    endif()
endif()

include(CTest)
if(AETHERI_POLICY_PROBES)
    add_executable(probe_policy tests/probe_policy.c)
    target_compile_definitions(probe_policy PRIVATE _GNU_SOURCE)
    target_compile_options(probe_policy PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(BUILD_TESTING)
    add_executable(test_fixture tests/test_fixture.c)
    target_compile_options(test_fixture PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(test_fixture PRIVATE aetheri_core)
    add_test(NAME core.fixture COMMAND Python3::Interpreter
             "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_fixture_native.py" $<TARGET_FILE:test_fixture> "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(core.fixture PROPERTIES TIMEOUT 40)
    add_executable(test_catalogue tests/test_catalogue.c)
    target_compile_definitions(test_catalogue PRIVATE _GNU_SOURCE)
    target_compile_options(test_catalogue PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_options(test_catalogue PRIVATE -Wl,--wrap=sqlite3_exec -Wl,--wrap=fsync)
    target_link_libraries(test_catalogue PRIVATE aetheri_catalogue)
    foreach(case IN ITEMS basic reject recovery accounts)
        add_test(NAME catalogue.${case} COMMAND test_catalogue ${case} "${CMAKE_CURRENT_BINARY_DIR}")
        set_tests_properties(catalogue.${case} PROPERTIES TIMEOUT 30)
    endforeach()
    add_test(NAME catalogue.interop COMMAND Python3::Interpreter
             "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_catalogue_native.py" $<TARGET_FILE:test_catalogue>)
    set_tests_properties(catalogue.interop PROPERTIES TIMEOUT 45)
    add_executable(test_core tests/test_core.c)
    target_compile_definitions(test_core PRIVATE _GNU_SOURCE)
    target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(test_core PRIVATE aetheri_core)
    add_test(NAME core.owner COMMAND test_core owner "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME core.storage COMMAND test_core storage "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(core.owner core.storage PROPERTIES TIMEOUT 10)
    if(AETHERI_LINUX_IPC)
        add_executable(test_peer tests/test_peer.c)
        target_compile_definitions(test_peer PRIVATE _GNU_SOURCE)
        target_compile_options(test_peer PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        target_link_options(test_peer PRIVATE -Wl,--wrap=sd_bus_query_sender_creds)
        target_link_libraries(test_peer PRIVATE aetheri_linux)
        add_test(NAME linux.peer COMMAND test_peer)
        set_tests_properties(linux.peer PROPERTIES TIMEOUT 30)
        add_executable(test_broker tests/test_broker.c src/authority/broker.c)
        target_compile_definitions(test_broker PRIVATE _GNU_SOURCE)
        target_compile_options(test_broker PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        target_link_libraries(test_broker PRIVATE aetheri_linux)
        target_link_options(test_broker PRIVATE -Wl,--wrap=ae_process_observe_registration -Wl,--wrap=ae_process_verify)
        add_test(NAME authority.broker COMMAND test_broker)
        set_tests_properties(authority.broker PROPERTIES TIMEOUT 25)
        if(AETHERI_PRIVILEGED_TESTS)
            add_executable(test_process tests/test_process.c)
            target_compile_definitions(test_process PRIVATE _GNU_SOURCE)
            target_compile_options(test_process PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
            target_link_libraries(test_process PRIVATE aetheri_linux)
            target_link_options(test_process PRIVATE -Wl,--wrap=ioctl)
            add_test(NAME linux.process COMMAND test_process)
            set_tests_properties(linux.process PROPERTIES TIMEOUT 30)
        endif()
    endif()
    if(AETHERI_PRIVILEGED_TESTS)
        add_test(NAME core.mount_containment COMMAND sudo -n unshare --mount --propagation private
                 $<TARGET_FILE:test_core> mount "${CMAKE_CURRENT_BINARY_DIR}")
        set_tests_properties(core.mount_containment PROPERTIES TIMEOUT 10)
    endif()
endif()
